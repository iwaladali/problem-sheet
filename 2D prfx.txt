vector<vector<ll>> prfx;

void buildPrefix(vector<vector<ll>>& v, int n, int m) {
    prfx.assign(n + 1, vector<ll>(m + 1, 0));

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++) {
            prfx[i][j] = v[i - 1][j - 1]
                       + prfx[i - 1][j]
                       + prfx[i][j - 1]
                       - prfx[i - 1][j - 1];
        }
    }
}

ll Query(int x1, int y1, int x2, int y2) {
    return prfx[x2][y2]
         - prfx[x1 - 1][y2]
         - prfx[x2][y1 - 1]
         + prfx[x1 - 1][y1 - 1];
}
int n, m;
cin >> n >> m;

vector<vector<ll>> v(n, vector<ll>(m));

for (int i = 0; i < n; i++)
    for (int j = 0; j < m; j++)
        cin >> v[i][j];

buildPrefix(v, n, m);

// مجموع المستطيل من (x1,y1) إلى (x2,y2)
cout << Query(x1, y1, x2, y2) << '\n';
