vector<int> next_smaller(vector<int>& a) {
    int n = a.size();
    vector<int> nxt(n, n);
    stack<int> st;

    for (int i = n - 1; i >= 0; i--) {
        while (!st.empty() && a[st.top()] >= a[i]) {
            st.pop();
        }

        if (!st.empty()) {
            nxt[i] = st.top();
        }

        st.push(i);
    }

    return nxt;
}
