
vector<int>par;
vector<ll> Dijkstra_s(int n,int start,int end,vector<vector<pair<int,int>>>adj) {
  vector<bool>vis(n+1);
  priority_queue<pair<ll, int>,vector<pair<ll, int>>, greater<pair<ll, int>>>pq;
  vector<ll>dis(n+1,__LONG_LONG_MAX__);
  dis[start]=0;
  pq.push({0,start});
  vis[start]=true;
  bool ok=false;
  while(!pq.empty()){
    pair<ll,ll>point=pq.top();
    vis[point.sc]=true;
    pq.pop();
    if(point.sc==end)
    break;
    if(point.fr>dis[point.sc]){      
      continue;
  }
      for(auto ch:adj[point.sc]){
        if(!vis[ch.fr])
        if((dis[ch.fr]>(dis[point.sc])+ch.sc))
       {
        par[ch.fr]=point.sc;
         dis[ch.fr]=(dis[point.sc])+ch.sc;
        pq.push({dis[point.sc]+ch.sc,ch.fr});
      }
      }
    
  }
  return dis;
  
}
