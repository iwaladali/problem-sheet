
struct Node
{
   ll  cost;
   int node;
   Node(ll c, int n){
      cost= c;
      node= n;
      
   }
 
   
};
 
struct Compare {
    bool operator()(const Node& a, const Node& b) const {
        return a.cost > b.cost;
    }
};
 
   
vector<int>par;
int n;
ll Dijkstra_s(int n,int start,int end, vector<vector<Node>>&adj) {
    par.resize(n+1);
    priority_queue<Node, vector<Node>, Compare>pq;  
 

    vector<ll>dist(n+1, inf);
  
    pq.push(Node(0, start));
 
    dist[start]=0;
 
    while(!pq.empty()){
    
    Node point=pq.top();
  
    pq.pop();
 
    if(point.node==end)
    return point.cost;
 
    if(point.cost>dist[point.node])    
      continue;
  
      for(auto ch:adj[point.node]){
                     
        int cost= point.cost+ch.cost;
 
        if (cost<dist[ch.node]){
              dist[ch.node]= cost ; 
              par[ch.node]=point.node;           
              pq.push(Node(dist[ch.node], ch.node));                        
      }
      }
    
  }
  return -1;
  
}
