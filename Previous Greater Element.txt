vector<int> prev_greater(vector<int>& a) {
    int n = a.size();
    vector<int> prv(n, -1);
    stack<int> st;

    for (int i = 0; i < n; i++) {
        while (!st.empty() && a[st.top()] <= a[i]) {
            st.pop();
        }

        if (!st.empty()) {
            prv[i] = st.top();
        }

        st.push(i);
    }

    return prv;
}
